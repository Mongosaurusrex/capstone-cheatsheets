\[
\begin{array}{*5{>{\displaystyle}l}}
\textbf{Dist} & \log p(x|\theta) & \ell'(\theta) & \ell''(\theta) & \textbf{MLE} \\ \hline
\text{Bern}(p) & x\log p+(1-x)\log(1-p) &
\frac{x}{p}-\frac{1-x}{1-p} &
-\frac{x}{p^2}-\frac{1-x}{(1-p)^2} &
\hat p=\bar x \\ \hline
\text{Bin}(n,p) & k\log p+(n-k)\log(1-p) &
\frac{k}{p}-\frac{n-k}{1-p} &
-\frac{k}{p^2}-\frac{n-k}{(1-p)^2} &
\hat p=\frac{k}{n} \\ \hline
\text{Geom}(p) & \log p+(k-1)\log(1-p) &
\frac1p-\frac{k-1}{1-p} &
-\frac1{p^2}-\frac{k-1}{(1-p)^2} &
\hat p=\frac1{\bar x} \\ \hline
\text{Pois}(\lambda) & x\log\lambda-\lambda &
\frac{x}{\lambda}-1 &
-\frac{x}{\lambda^2} &
\hat\lambda=\bar x \\ \hline
\text{Exp}(\lambda) & \log\lambda-\lambda x &
\frac1\lambda-x &
-\frac1{\lambda^2} &
\hat\lambda=\frac1{\bar x} \\ \hline

-\frac{(x-\mu)^2}{2\sigma^2}-\frac12\log\sigma^2 &

-\frac1{\sigma^2} &
\hat\mu=\bar x

\section{Conditioning, independence, and counting}

\subsection{Conditional probability}

\[
\Pr\!\left(\bigcap_{i=1}^n A_i\right)
= \Pr(A_1)\prod_{i=2}^n \Pr\!\left(A_i \mid \bigcap_{j=1}^{i-1} A_j\right)
\]

\[
\Pr(B)=\sum_i \Pr(A_i)\Pr(B\mid A_i),\quad
\Pr(A_i\mid B)=\frac{\Pr(B\mid A_i)\Pr(A_i)}{\sum_j \Pr(B\mid A_j)\Pr(A_j)}
\]

\[
\Pr(A\mid B)=
\frac{\Pr(B\mid A)\Pr(A)}
{\Pr(B\mid A)\Pr(A)+\Pr(B\mid A^c)\Pr(A^c)}
\]

\subsection{Independence}

\[
\Pr\!\left(\bigcap_{i\in S} A_i\right)=\prod_{i\in S}\Pr(A_i)
\]

\begin{itemize}
\item $A\perp B \iff \Pr(A\mid B)=\Pr(A)$
\item Independence $\not\Rightarrow$ conditional independence
\item $X\perp Y \Rightarrow \E[g(X)h(Y)]=\E[g(X)]\E[h(Y)]$
\end{itemize}

\subsection{Counting}
\begin{itemize}
\item Permutations: $n!$
\item $k$-permutations: $n!/(n-k)!$
\item Combinations: $n!/(k!(n-k)!)$
\item Partitions: $n!/(n_1!\cdots n_r!)$
\end{itemize}

\section{Random variables}

\subsection{Expectation and variance}

\[
\E[aX+bY]=a\E[X]+b\E[Y],\quad
\E[X]=\E[\E[X\mid Y]]=\int \E[X\mid Y=y]f_Y(y)\,dy
\]

\[
\Var(X)=\E[X^2]-(\E[X])^2,\quad
\Var(cX)=c^2\Var(X)
\]

\[
\Cov(X,Y)=\E[XY]-\E[X]\E[Y],\quad
\Var(X+Y)=\Var(X)+\Var(Y)+2\Cov(X,Y)
\]

\[
\Var(X)=\E[\Var(X\mid Y)]+\Var(\E[X\mid Y]),\quad
X\perp Y \Rightarrow \Cov(X,Y)=0
\]

\subsection{Conditioning rules}

\[
\E[g(Y)X\mid Y]=g(Y)\E[X\mid Y],\quad
\Var(X\mid Y)=\E[X^2\mid Y]-(\E[X\mid Y])^2
\]

\[
X\perp Y \Rightarrow
\E[X\mid Y]=\E[X],\quad
\Var(X\mid Y)=\Var(X)
\]
